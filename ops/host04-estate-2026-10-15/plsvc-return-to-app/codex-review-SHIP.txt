// coexist while one bundle cannot be minted twice (Codex pass 4). A schema
// change, not this packet's.
const _mintQueues = new Map<string, Promise<unknown>>()
async function serialisedPerAccount<T>(liteAccountId: string, fn: () => Promise<T>): Promise<T> {
  const prev = _mintQueues.get(liteAccountId) ?? Promise.resolve()
  const run = prev.catch(() => undefined).then(fn)
  _mintQueues.set(liteAccountId, run)
  try {
    return await run
  } finally {
    if (_mintQueues.get(liteAccountId) === run) _mintQueues.delete(liteAccountId)
  }
}

export async function reuseOrMintPaymentLink(
  input: MintPaymentLinkInput & { liteAccountId: string },

  })

//      payment �?" belt-and-braces, do not remove this check.
//   5. A link whose liteAccountId is RETIRED is refused BEFORE fiservCharge and
//      canceled (2026-09-15). The credit-side guard in assertMagnusOwnership
//      is not enough here: it would fire after the card was charged.
//
// CONCURRENCY: the only real double-charge guard is the atomic claim below
// (updateMany pending->processing gated on count===1). The status checks that
// follow it are just for a friendly, accurate error message �?" do not rely on
// a preceding findUnique() read to decide whether to charge; two concurrent
// requests would both pass a plain read-then-write check.
//
// WA TOP-UP WALLET CREDIT (added alongside WA-top-up P-link wiring):
//   When a link carries liteAccountId (set by the WhatsApp top-up Flow
//   handler at mint time), a successful payment must ALSO credit the
//   customer's Magnus wallet �?" this route is the only hop that knows the
//   charge genuinely succeeded, so it owns firing the credit. Mirrors the
//   MOTO precedent in app/api/platform/fiserv/charge/route.ts:
//     resolve liteAccountId -> magnusUserId -> addMagnusCredit -> ledger row
//     (idempotency key 'plink:<token>') -> fire-and-forget WA confirmation.
//   A credit failure must NEVER unmark the link as paid �?" the charge already
//   settled with the processor. Failures are logged loudly and surfaced in
//   the response as creditError for operator reconciliation.
//
// WA AIRTIME TOP-UP (Reloadly) �?" same seam, different purpose:
//   When a link carries reloadlyRecipientPhone (set by the WA airtime
//   concierge branch at mint time instead of relying on liteAccountId alone),
//   a successful payment fires a real Reloadly send instead of a Magnus
//   credit. Same non-masking rule: a failed send never unmarks the link as
//   paid �?" the charge already settled. Idempotency key 'plink-airtime:<token>'
//   (distinct from the Magnus path's 'plink:<token>') on the same
//   liteMinuteLedger table, checked-then-set before calling Reloadly so a
//   retried/replayed charge attempt (already blocked by the atomic claim
//   above) can never fire a second send even if this block were ever reached
//   twice. Failures surfaced as topupError, never silently dropped.
import { NextRequest, NextResponse } from 'next/server'
import { validateInternalSecret, validatePayPageSecret } from '@/app/lib/internal-auth'
import { prisma }                    from '@/app/lib/prisma'
import { fiservCharge }              from '@/app/lib/payments/fiserv'
import { recordFiservCharge }        from '@/app/lib/payments/fiserv-charge-log'
import { getMagnusUserById } from '@/app/lib/magnus'
import { creditWalletOnce } from '@/app/lib/wallet-credit'
import { sendTopup }                 from '@/app/lib/reloadly-client'
import { sendWhatsAppMessage, sendWhatsAppTemplate } from '@/app/lib/whatsapp'
import { isRetiredLiteAccount, recordRetiredMoneyRefusal } from '@/app/lib/lite-retire'
import { isLegacyBonusPricedCardLink } from '@/app/lib/lite-topup'
import { airtimeSendAvailable, airtimeUnavailable } from '@/app/lib/lite-airtime'

export const dynamic = 'force-dynamic'

// Fixed, honest, customer-facing copy for a genuine card decline. Never show
// the customer raw/technical gateway text �?" that detail (result.error) is for
// operators only (lastError column + recordFiservCharge), not the browser.
// Used ONLY when the gateway actually reported a decline-shaped result
// (a status was returned, or the ok-without-ref incident case) �?" NOT for
// config/currency/network errors that never reached a card at all; those
// keep their own already-plain-English result.error message.
const DECLINE_MESSAGE_FOR_CUSTOMER = 'Your card was declined �?" no charge was made. Try another card.'

export async function POST(
  req: NextRequest,
  { params }: { params: Promise<{ token: string }> }
) {
  if (!(validateInternalSecret(req) || validatePayPageSecret(req))) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  const { token } = await params

  let body: Record<string, unknown>
  try {
    body = (await req.json()) as Record<string, unknown>
  } catch {
    return NextResponse.json({ error: 'Invalid JSON body' }, { status: 400 })
  }

  const { card, payer } = body as {
    card?:  { cardNumber?: string; expMonth?: string; expYear?: string; cvv?: string }
    payer?: { name?: string; email?: string; address?: string }
  }

  if (!card?.cardNumber || !card?.expMonth || !card?.expYear || !card?.cvv) {
    return NextResponse.json({ error: 'card.cardNumber, expMonth, expYear, cvv are required' }, { status: 400 })
  }
  if (!payer?.name || !payer?.email || !payer?.address) {
    return NextResponse.json({ error: 'payer.name, email, address are required' }, { status: 400 })
  }

  const existing = await prisma.paymentLink.findUnique({ where: { token } })
  if (!existing) {
    return NextResponse.json({ error: 'not_found' }, { status: 404 })
  }

  // Atomic claim: the DB only lets one request flip pending -> processing.
  // This �?" not the reads above or below �?" is what actually kills the race.
  const claim = await prisma.paymentLink.updateMany({
    where: { token, status: 'pending' },
    data:  { status: 'processing' },
  })

  if (claim.count !== 1) {
    const current = await prisma.paymentLink.findUnique({ where: { token } })
    if (!current) {
      return NextResponse.json({ error: 'not_found' }, { status: 404 })
    }
    if (current.status === 'paid') {
      return NextResponse.json({ ok: false, status: 'ALREADY_PAID', error: 'This link has already been paid.' }, { status: 409 })
    }
    if (current.status === 'processing') {
      return NextResponse.json({ ok: false, status: 'IN_PROGRESS', error: 'A charge is already being processed for this link.' }, { status: 409 })
    }
    if (current.status === 'canceled') {
      return NextResponse.json({ ok: false, status: 'CANCELED', error: 'This link was canceled after too many failed attempts.' }, { status: 410 })
    }
    return NextResponse.json({ ok: false, status: 'EXPIRED', error: 'This payment link has expired.' }, { status: 410 })
  }

  // We now hold the exclusive 'processing' claim on this link.
  if (existing.expiresAt.getTime() < Date.now()) {
    // Claimed an already-expired link �?" release it without counting a failed attempt.
    await prisma.paymentLink.update({ where: { token }, data: { status: 'expired' } })
    return NextResponse.json({ ok: false, status: 'EXPIRED', error: 'This payment link has expired.' }, { status: 410 })
  }

  // Any of the three Reloadly fields present routes away from the ordinary
  // Magnus-credit branch �?" keyed on "any", not just reloadlyRecipientPhone,
  // so a row that somehow ends up with only a PARTIAL set (manual DDL/DML,
  // a future writer bypassing mintPaymentLink's all-or-none validation)
  // fails loud in the branch below instead of silently taking a Magnus
  // credit for what was actually an airtime purchase. (Hoisted 2026-09-25 so
  // the pre-charge wallet check keys on the same predicate.)
  const hasAnyReloadlyField = existing.reloadlyOperatorId != null || existing.reloadlyRecipientPhone != null || existing.reloadlyFaceAmountUsd != null

  // A RETIRED ACCOUNT NEVER REACHES fiservCharge (2026-09-15, PKT-RETIREMENT-
  // STOPS-MONEY). This is the check that matters: the ownership assertion
  // inside addMagnusCredit also refuses a retired row, but that runs AFTER the
  // card has been charged �?" money taken, nothing delivered, which is exactly
  // what happened twice on �?�252k on 2026-09-06 (links minted before the
  // retirement, charged hours after it). So the row is read here, before the
  // processor is contacted, and the link is CANCELED �?" not left 'pending' �?"
  // so a retry cannot find it again. Not counted as a failed attempt: the
  // card was never tried. Links minted for tenants (no liteAccountId) are
  // unaffected.
  const refuseRetired = async (owner: { id: string }) => {
    console.warn(`[payment-link/charge] REFUSED before charge: token=${token} liteAccount=${owner.id} account_retired �?" link canceled`)
    // If this update fails the link stays 'processing', which is itself
    // fail-closed (a processing link answers IN_PROGRESS to every later
    // charge and can never reach the processor); the 410 still goes out.
    await prisma.paymentLink
      .update({ where: { token }, data: { status: 'canceled', lastError: 'account_retired' } })
      .catch((e: any) => console.error(`[payment-link/charge] cancel-on-retired failed for token=${token}: ${e?.message} (link left processing �?" no charge is possible in that state)`))
    await recordRetiredMoneyRefusal({ source: 'fiserv:payment-link/charge', liteAccountId: owner.id, amount: Number(existing.amount), currency: existing.currency, ref: token, detail: 'link canceled before fiservCharge; no money taken' })
    return NextResponse.json(
      { ok: false, status: 'ACCOUNT_RETIRED', error: 'This line is no longer in service �?" no charge was made.' },
      { status: 410 },
    )
  }
  if (existing.liteAccountId) {
    const owner = await prisma.liteAccount.findUnique({
      where: { id: existing.liteAccountId },
      select: { id: true, userId: true, ownerPhone: true, magnusUserId: true },
    })
    // Only a RETIRED row is refused here; a missing row on a wallet-credit
    // link is refused just below (no wallet), before the charge.
    if (owner && isRetiredLiteAccount(owner)) return refuseRetired(owner)

    // A WALLET THAT CANNOT BE CREDITED NEVER REACHES fiservCharge either
    // (2026-09-25, def-payment-link-charge-captures-card-before-verifying-
    // wallet-2026-09-25). A wallet-credit link (liteAccountId and no Reloadly
    // field -- the SAME predicate the credit branch below uses) whose account
    // has no magnusUserId, or whose Magnus user getMagnusUserById cannot find,
    // used to be charged first and then logged 'no-magnus-user': money taken,
    // nothing credited. Checked here with the same helpers the Personal Line
    // top-up route uses, before the processor is contacted; not counted as a
    // failed attempt (the card was never tried). Airtime and tenant links are
    // not touched.
    //   - no magnusUserId on the row (or no row): definitive -> link CANCELED
    //     with lastError, exactly as a retired account is;
    //   - getMagnusUserById null: no such Magnus user -- but that helper also
    //     answers null when Magnus is unreachable, so the link is released to
    //     'pending' (payable later), not canceled;
    //   - getMagnusUserById throws: unknown -> released to 'pending', 503.
    // Fail closed in every case: nothing is charged.
    if (!hasAnyReloadlyField) {
      let wallet: 'ok' | 'none' | 'not_found' | 'unverifiable' = 'ok'
      if (!owner?.magnusUserId) {
        wallet = 'none'
      } else {
        try {
          if (!(await getMagnusUserById(owner.magnusUserId))) wallet = 'not_found'
        } catch {
          wallet = 'unverifiable'
        }
      }
      // The probe above awaited Magnus (up to its timeout): the account can be
      // RETIRED or DELETED meanwhile (Codex on #236; fixture resets hard-delete
      // lite_accounts). Re-read it right before the charge -- nothing else
      // awaits between here and fiservCharge: retired -> refused exactly as
      // above; no row -> the definite no-wallet refusal below.
      if (wallet === 'ok') {
        const fresh = await prisma.liteAccount.findUnique({
          where: { id: existing.liteAccountId },
          select: { id: true, userId: true, ownerPhone: true },
        })
        if (fresh && isRetiredLiteAccount(fresh)) return refuseRetired(fresh)
        if (!fresh) wallet = 'none'
      }
      if (wallet !== 'ok') {
        const lastError = wallet === 'unverifiable' ? 'wallet_unverifiable' : 'no_wallet'
        console.warn(`[payment-link/charge] REFUSED before charge: token=${token} liteAccount=${existing.liteAccountId} wallet=${wallet} �?" no charge made`)
        await prisma.paymentLink
          .update({ where: { token }, data: wallet === 'none' ? { status: 'canceled', lastError } : { status: 'pending', lastError } })
          .catch((e: any) => console.error(`[payment-link/charge] release-on-${lastError} failed for token=${token}: ${e?.message} (link left processing �?" no charge is possible in that state)`))
        if (wallet === 'unverifiable') {
          return NextResponse.json(
            { ok: false, status: 'WALLET_UNVERIFIABLE', error: 'We could not confirm the wallet for this line right now �?" no charge was made. Please try again in a few minutes.' },
            { status: 503 },
          )
        }
        return NextResponse.json(
          { ok: false, status: 'NO_WALLET', error: 'This line has no wallet we can top up �?" no charge was made. Please contact support.' },
          { status: wallet === 'none' ? 410 : 409 },
        )
      }
    }
  }

  // A bundle link minted BEFORE the price correction carries the bonus-
  // inclusive figure as its charge (27.50 for "EC$25", 60.00 for "EC$50").
  // The owner ruled EC$25 charges EC$25; that must hold for every charge made
  // after the fix ships, and the row �?" not the code �?" decides the figure, so
  // the fix in the minters alone would still let an old link charge the old
  // price for up to its 24h TTL. Refuse it, exactly as an expired link is
  // refused: released (not a failed attempt), never charged, and the customer
  // is told to ask for a fresh link. See isLegacyBonusPricedCardLink().
  if (isLegacyBonusPricedCardLink(existing)) {
    await prisma.paymentLink.update({ where: { token }, data: { status: 'expired', lastError: 'superseded: minted at the pre-2026-09-15 bonus-inclusive price' } })
    return NextResponse.json({ ok: false, status: 'PRICE_SUPERSEDED', error: 'This payment link was issued at an old price and cannot be charged �?" no charge was made. Please request a new top-up link.' }, { status: 410 })
  }

  // AN AIRTIME LINK THAT CANNOT DELIVER NEVER REACHES fiservCharge (Codex on
  // #237). The minters (WhatsApp concierge, app /api/lite/airtime/pay-link)
  // run this preflight when the link is MINTED, but the card can be charged up
  // to the link's TTL later -- after the send gate was closed or the Reloadly
  // float spent -- and sendTopup() below runs only AFTER capture: money taken,
  // no airtime. The same shared preflight runs here, the last await before the
  // processor. The link is released to 'pending' (not canceled, not an
  // attempt): the condition is transient and the customer can pay later.
  // Only a COMPLETE airtime link: a partial Reloadly field set keeps its own
  // fail-loud handling below, unchanged.
  if (existing.liteAccountId && existing.reloadlyRecipientPhone && existing.reloadlyOperatorId != null && existing.reloadlyFaceAmountUsd != null) {
    // The float check is not a reservation (Codex on #237, 2nd pass): two
    // charges racing for the last of the float would both pass and the second
    // would capture with no airtime. Every OTHER airtime link currently
    // 'processing' (claimed, not yet captured -- sendTopup follows capture
    // within seconds) is counted against the float first. Concurrent charges
    // that together exceed it both refuse, uncharged, rather than one being
    // charged for nothing. This narrows the race; it is not a hard reservation.
    const inFlight = await prisma.paymentLink.findMany({
      where:  { status: 'processing', token: { not: token }, reloadlyFaceAmountUsd: { not: null } },
      select: { reloadlyFaceAmountUsd: true },
    })
    const reservedUsd = inFlight.reduce((sum: number, l: { reloadlyFaceAmountUsd: unknown }) => sum + Number(l.reloadlyFaceAmountUsd), 0)
    const airtime = await airtimeSendAvailable(Number(existing.reloadlyFaceAmountUsd), reservedUsd)
    if (airtimeUnavailable(airtime)) {
      console.warn(`[payment-link/charge] REFUSED before charge: token=${token} airtime ${airtime.reason} �?" no charge made`)
      await prisma.paymentLink
        .update({ where: { token }, data: { status: 'pending', lastError: `airtime_unavailable:${airtime.reason}` } })
        .catch((e: any) => console.error(`[payment-link/charge] release-on-airtime_unavailable failed for token=${token}: ${e?.message} (link left processing �?" no charge is possible in that state)`))
      return NextResponse.json(
        { ok: false, status: 'AIRTIME_UNAVAILABLE', error: 'Airtime top-ups are not available right now �?" no charge was made. Please try again later.' },
        { status: 503 },
      )
    }
  }

  const last4 = card.cardNumber.replace(/\D/g, '').slice(-4)

  // Amount/currency come from the link row �?" never from the client. `existing`
  // was read immediately before the atomic claim succeeded, and nothing else
  // can mutate a 'pending' row except this same claim, so it is still fresh.
  //
  // amount is a Prisma Decimal �?" convert explicitly to a plain number for the
  // two call sites that need one (Fiserv's JSON payload and the local charge
  // log). Fiserv itself is sent major-currency-unit decimal (e.g. 25.00), not
  // cents; see fiserv.ts FiservChargeInput.amount doc comment.
  const amountNumber = Number(existing.amount)

  const result = await fiservCharge({
    cardNumber:   card.cardNumber,
    expMonth:     card.expMonth,
    expYear:      card.expYear,
    cvv:          card.cvv,
    amount:       amountNumber,
    currency:     existing.currency,
    payerName:    payer.name,
    payerEmail:   payer.email,
    payerAddress: payer.address,
  })

  // Belt-and-braces (2026-07-07T00:15Z incident): a result is only a genuine
  // payment when it is ok AND carries a transaction ref. fiserv.ts now enforces
  // this upstream, but this route does not trust `result.ok` alone �?" an
  // ok-without-ref result is treated as a failure here too, never as 'paid'.
  const isGenuinePayment = result.ok && Boolean(result.ref)

  // A "genuine decline" is a result the gateway actually adjudicated on the
  // card (it returned a status), or the incident case (ok-without-ref, which
  // fiserv.ts now also flags with a status when it can). Distinguish this
  // from config/network/currency failures, which never reached a card and
  // should keep their own accurate, already-plain-English message rather than
  // falsely telling the customer their card was declined.
  const isGenuineDecline = !isGenuinePayment && !result.disabled && Boolean(result.status)

  recordFiservCharge({
    accountId: existing.tenantId ?? undefined,
    amount:    amountNumber,
    currency:  existing.currency,
    last4,
    status:    result.status ?? (result.disabled ? 'disabled' : isGenuinePayment ? 'ok' : 'error'),
    ref:       result.ref,
    ok:        isGenuinePayment,
    createdBy: existing.createdBy ?? undefined,
  })

  let attempts = existing.attempts
  let canceledNow = false

  // Populated only on the genuine-payment branch below, when the link
  // carries a liteAccountId �?" surfaced in the response for operator visibility.
  // balanceAfter is null when the post-credit readback failed -- the credit
  // still succeeded, we just refuse to name a balance we did not confirm.
  let credited: { ecd: number; magnusUserId: string; balanceAfter: number | null } | undefined
  let creditError: string | undefined

  // Populated only on the genuine-payment branch below, when the link
  // carries a reloadlyRecipientPhone �?" surfaced in the response for operator
  // visibility. Mutually exclusive with credited/creditError: a link is
  // either a Magnus-credit top-up or a Reloadly-airtime top-up, never both.
  let topup: { transactionId: string; deliveredAmount: number; deliveredAmountCurrency: string } | undefined
  let topupError: string | undefined

  if (isGenuinePayment) {
    await prisma.paymentLink.update({
      where: { token },
      data:  { status: 'paid', transactionId: result.ref, paidAt: new Date() },
    })

    // �"?�"? WA top-up wallet credit (the missing hop) �"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?
    // Only fires when this link was minted with a liteAccountId (the WA
    // top-up flow does this at mint time; agent-minted Flow-A links for
    // arbitrary invoices do not, and are unaffected).
    if (existing.liteAccountId && existing.reloadlyRecipientPhone) {
      // �"?�"? Airtime top-up (Reloadly) �?" WA "top up my phone" seam �"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?�"?
      // Checked FIRST and separately from the Magnus-credit branch below: a
      // link minted for airtime never falls through to a Magnus credit, even
      // if liteAccountId is also set (it always is, for WA confirmation).
      try {
        if (existing.reloadlyOperatorId == null || existing.reloadlyFaceAmountUsd == null) {
          console.error(`[payment-link/charge] token=${token} has reloadlyRecipientPhone but is missing reloadlyOperatorId/reloadlyFaceAmountUsd �?" cannot send`)
          topupError = 'missing-operator'
        } else {
          const idempotencyRef = `plink-airtime:${token}`
          const alreadyProcessed = await prisma.liteMinuteLedger.findFirst({
            where: { stripeSessionId: idempotencyRef },
          })

          if (alreadyProcessed) {
            console.log(`[payment-link/charge] airtime already sent for token ${token} �?" skipping (idempotent replay)`)
            const meta = (alreadyProcessed.meta as any) ?? {}
            topup = {
              transactionId:           meta.transactionId ?? '',
              deliveredAmount:         meta.deliveredAmount ?? 0,
              deliveredAmountCurrency: meta.deliveredAmountCurrency ?? '',
            }
          } else {
            try {
              // FACE amount, not amountNumber (the Fiserv charge, which
              // includes margin) �?" sending the marked-up price to Reloadly
              // either gets rejected outright (FIXED-denomination operators)
              // or silently overspends EPIC's float and over-delivers to the
              // recipient (RANGE operators), in both cases breaking the
              // amount the confirm message already promised the customer.
              const faceAmount = Number(existing.reloadlyFaceAmountUsd)
              const receipt = await sendTopup({
                operatorId:       existing.reloadlyOperatorId,
                amount:           faceAmount,
                recipientPhone:   existing.reloadlyRecipientPhone,
                customIdentifier: idempotencyRef,
              })

              topup = {
                transactionId:           receipt.transactionId,
                deliveredAmount:         receipt.deliveredAmount,
                deliveredAmountCurrency: receipt.deliveredAmountCurrency,
              }

              try {
                await prisma.liteMinuteLedger.create({
                  data: {
                    liteAccountId:   existing.liteAccountId,
                    type:            'topup',
                    minutes:         0,
                    stripeSessionId: idempotencyRef,
                    meta: {
                      source:                  'wa_plink_airtime',
                      recipientPhone:          existing.reloadlyRecipientPhone,
                      operatorId:              existing.reloadlyOperatorId,
                      transactionId:           receipt.transactionId,
                      deliveredAmount:         receipt.deliveredAmount,
                      deliveredAmountCurrency: receipt.deliveredAmountCurrency,
                      faceAmountUsd:           faceAmount,
                      chargedAmount:           amountNumber,
                      chargedCurrency:         existing.currency,
                      token,
                    },
                  },
                })
              } catch (ledgerErr: any) {
                // Race-condition duplicate write �?" send already happened; log and continue.
                console.warn(`[payment-link/charge] airtime ledger write failed (possible race): ${ledgerErr.message}`)
              }

              console.log(`[payment-link/charge] airtime sent -> +${existing.reloadlyRecipientPhone} txn=${receipt.transactionId} token=${token}`)

              // Fire-and-forget, best-effort �?" wrapped so nothing here (sync or
              // async) can ever propagate and mask the already-settled charge.
              try {
                const laForConfirm = await prisma.liteAccount.findUnique({
                  where:  { id: existing.liteAccountId },
                  select: { ownerPhone: true },
                }).catch(() => null)
                if (laForConfirm?.ownerPhone) {
                  Promise.resolve(sendWhatsAppMessage(
                    laForConfirm.ownerPhone,
                    `Airtime sent: +${existing.reloadlyRecipientPhone} received ${receipt.deliveredAmount} ${receipt.deliveredAmountCurrency}. Ref: ${receipt.transactionId}`,
                  )).catch((e: any) =>
                    console.error(`[payment-link/charge] WA airtime confirmation failed (non-fatal) for token=${token}:`, e?.message),
                  )
                }
              } catch (notifyEx: any) {
                console.error(`[payment-link/charge] airtime WA confirmation setup failed (non-fatal) for token=${token}:`, notifyEx?.message)
              }
            } catch (sendEx: any) {
              // Charge already settled above (status='paid') �?" never mask a
              // captured payment by reopening/reverting it. Log loudly, surface
              // topupError, let the operator reconcile or manually retry the
              // send with the same idempotencyRef.
              console.error(`[payment-link/charge] sendTopup FAILED token=${token} recipient=${existing.reloadlyRecipientPhone}: ${sendEx.message}`)
              topupError = 'reloadly-send-failed'

              try {
                const laForConfirm = await prisma.liteAccount.findUnique({
                  where:  { id: existing.liteAccountId },
                  select: { ownerPhone: true },
                }).catch(() => null)
                if (laForConfirm?.ownerPhone) {
                  Promise.resolve(sendWhatsAppMessage(
                    laForConfirm.ownerPhone,
                    `Payment received, but airtime delivery failed �?" support will follow up. No second charge will be made.`,
                  )).catch(() => {})
                }
              } catch { /* best-effort notification only �?" topupError above is already set */ }
            }
          }
        }
      } catch (topupEx: any) {
        // Unexpected error in the airtime block �?" charge already settled
        // above; log for reconciliation, do not touch link status.
        console.error(`[payment-link/charge] airtime block threw (token=${token}, ref=${result.ref}):`, topupEx.message)
        topupError = 'reloadly-send-exception'
      }
    } else if (existing.liteAccountId && !hasAnyReloadlyField) {
      try {
        const la = await prisma.liteAccount.findUnique({
          where:  { id: existing.liteAccountId },
          select: { id: true, magnusUserId: true, ownerPhone: true, userId: true },
        })

        if (!la?.magnusUserId) {
          console.error(`[payment-link/charge] liteAccountId=${existing.liteAccountId} has no magnusUserId �?" cannot credit wallet (token=${token})`)
          creditError = 'no-magnus-user'
        } else {
          // existing.amount is what the CARD was charged, in the link's own
          // currency. creditAmountEcd is what the WALLET receives, pinned at
          // mint time by EVERY bundle minter since 2026-09-15 (PKT-TOPUP-
          // PRICE-CORRECTION): XCD bundles pin creditAmount (25 �+' 27.50,

/** The form POST came from a page on THIS host (Origin, else Sec-Fetch-Site). */
function sameOriginPost(req: NextRequest): boolean {
  const host = req.headers.get('x-forwarded-host') ?? req.headers.get('host') ?? req.nextUrl.host
  const origin = req.headers.get('origin')
  if (origin) {
    try { return new URL(origin).host === host } catch { return false }
  }
  return req.headers.get('sec-fetch-site') === 'same-origin'
}

/**

 * launch-blocking): def-wa-inapp-webview-hangs-on-go-link-2026-09-28.
 *
 * Real-device measurement: WhatsApp's own Android in-app browser GETs this

 * tap never reaches the server. Root cause is WA4A's WebView itself, not
 * this file's CSP -- htmlPrivate()'s `form-action 'none'` only ever applies
 * to launchHandoffPage(), which is reached AFTER this POST already
 * succeeded, so it cannot be what's blocking the POST that never fires.
 *
 * looksLikeHumanNavigation()'s NON_HUMAN_UA blocklist does not catch this:
 * it blocks the literal word "WhatsApp", but WA4A's real UA carries
 * "WA4A/<version>", never that word. That is deliberate on this function's
 * part too: this is a REAL human tap, not a bot, so it must not get
 * previewPage() (credential-free but also *helpless* -- no path forward).
 * It gets routed OUT to a real browser instead, before ANY OTP/chat-tap/
 * handoff machinery runs -- never racing it, never duplicating it.

// The only two vhosts nginx actually proxies /go/ from today (production and
// test.epic.dm's mirrored same-origin path -- see their sites-enabled
// configs). Codex review, P2: X-Forwarded-Host is client-suppliable through a
// misconfigured/permissive proxy; this page builds a URL carrying the
// activation nonce into an href, so �?" unlike sameOriginPost()'s use of the
// same header just to COMPARE against Origin �?" it must never trust an
// unvalidated host to CONSTRUCT one. Falls back to the real production host
// rather than trusting anything outside this allowlist.
const KNOWN_GO_HOSTS = new Set(['bff.epic.dm', 'test.epic.dm'])
function trustedGoHost(req: NextRequest): string {
  const claimed = req.headers.get('x-forwarded-host') ?? req.headers.get('host') ?? ''
  return KNOWN_GO_HOSTS.has(claimed) ? claimed : 'bff.epic.dm'
}

/** Resolve the wa.me number for a campaign link's owning Agent, same field
 *  preference used by tenant/[id]/agents and tenant/[id]/channels routes. */
function resolveWaNumber(agent: {
  whatsappNumber: string | null
  waDisplayNumber: string | null
  didNumber: string | null
  phoneNumber: string | null
}): string | null {
  const raw = agent.whatsappNumber ?? agent.waDisplayNumber ?? agent.didNumber ?? agent.phoneNumber
  if (!raw) return null
  const digits = raw.replace(/\D/g, '')
  return digits || null
}

export async function GET(_req: NextRequest, { params }: { params: Promise<{ code: string }> }) {
  const { code } = await params

  // 1. Campaign link lookup (new �?" agent-full-surface).
  const link = await prisma.agentCampaignLink.findUnique({ where: { code } }).catch(() => null)
  if (link) {
    const agent = await prisma.agent
      .findUnique({
        where: { id: link.agentId },
        select: { whatsappNumber: true, waDisplayNumber: true, didNumber: true, phoneNumber: true },
      })
      .catch(() => null)

    await prisma.agentCampaignLink
      .update({ where: { id: link.id }, data: { clickCount: { increment: 1 } } })
      .catch(() => null)

    const number = agent ? resolveWaNumber(agent) : null
    if (!number) {
      return NextResponse.json({ error: 'campaign-link-agent-has-no-whatsapp-number' }, { status: 404 })
    }

    const waUrl = link.contextSeed
      ? `https://wa.me/${number}?text=${encodeURIComponent(link.contextSeed)}`
      : `https://wa.me/${number}`

    return NextResponse.redirect(waUrl, 302)
  }

  // 1b. Sign-in short link (2026-09-02).
  //
  // The "open my app" URL in the WhatsApp welcome message was ~130 characters
  // of base64url sitting raw in a chat bubble �?" unreadable, wrapping over
  // several lines, and looking like something you should not tap. This route
  // was already the estate's short-link shape; it just only knew how to carry
  // activation nonces. See wa-signin-shortlink.ts for what is stored (the
  // phone, never a token) and why the code is sized as a credential.
  //
  // Ordered after campaign links and before activation nonces purely to match
  // the existing "cheapest lookup first" shape; the three code spaces have
  // different lengths and separate stores, so no code can resolve as two
  // things.
  const signinUrl = await resolveSigninShortCode(code)
  if (signinUrl) return NextResponse.redirect(signinUrl, 302)

  // 2. Lite Cloud Softphone activation nonce flow �?" WIRED 2026-08-18.
  //
  // activationUnavailablePage() is now the FLAG-OFF BRANCH ONLY, not a
  // hardcode. Between 2026-06-20 (326134b5 orphaned the secure path) and today,
  // this branch returned "unavailable" unconditionally, which is why the secure
  // XML endpoint had never succeeded once in 40 attempts �?" verifyActivationOtp
  // had zero callers, so nothing could ever set verified=true.
  //
  // THE RESPONSE IS DELIBERATELY IDENTICAL for a disabled account, an unknown
  // code and an expired nonce. That keeps this route from being an oracle for
  // whether a given nonce exists �?" the same reason the flag-off page never
  // mentions why it is unavailable.
  // NOT A PERSON: no code, no write, no oracle. Link-preview crawlers and
  // prefetchers land here. Measured 2026-09-15: WhatsApp's fetcher sent the
  // owner's OTP before he had opened anything.
  if (!looksLikeHumanNavigation(_req.headers)) return previewPage()

  // DIAGNOSIS CORRECTED 2026-09-29 (def-app-signin-stuck-in-whatsapp-inapp-
  // browser-regression-2026-09-29): #258 forced EVERY in-app UA straight to
  // inAppBrowserEscapePage() here, before chatActivatePage/otpPage ever
  // rendered, on the premise that WA4A's WebView could never complete the
  // POST those pages depend on. Real device logs proved that premise wrong:
  // the POST succeeded 9 times for normal WhatsApp (main app) across 15-28
  // Sep on 3 phone models. The actual failures were Samsung Dual Messenger
  // (a second, cloned WhatsApp instance with its own broken WebView) on one
  // test number -- #258 cost every normal customer an extra tap for a bug
  // that was never theirs. Reverted: in-app UAs render the same pages a
  // real browser gets. What stays: stuckFallbackSection() below, appended
  // to chatActivatePage/otpPage, for the rare case the POST really can't
  // fire.
  const uaHeader = _req.headers.get('user-agent')

  // The account this browser proved at signup, if any. Passed through so
  // beginActivation can skip a redundant second OTP �?" and ONLY when it matches
  // this nonce's own account. See lite-signup-session.ts.
  const sessionUserId = readSignupSession(_req.cookies.get(SIGNUP_SESSION_COOKIE)?.value)
  const bound = handoffCookieProves(_req.cookies.get(HANDOFF_COOKIE)?.value, code)
  const begun = await beginActivation(code, false, sessionUserId).catch(() => null)
  if (!begun || begun.state === 'expired') return activationUnavailablePage()

  // Per-account gate, re-checked at RENDER time and not only at mint time: a
  // link minted for the canary must stop working if the canary is removed from
  // the allowlist, otherwise the allowlist only scopes issuance and not access.
  if (!(await activationEnabledFor(begun.userId ?? begun.sipUsername))) return activationUnavailablePage()

  if (begun.state === 'verified') {
    // THIS browser typed the code (or proved the account at signup, in which
    // case beginActivation just verified it and the cookie is set below):
    // straight to the hand-off. A re-open after a store trip lands here.
    // Re-open by the browser that already holds this link's binding cookie.
    if (bound) return handoff(code, begun, _req, { setCookie: false })
    // CHAT-VERIFIED link (owner, 2026-09-25): no OTP -- but a GET NEVER
    // consumes, binds or reveals (Codex #231, P1). WhatsApp's link-preview
    // fetcher and safe-link scanners GET links posted in chat, and
    // looksLikeHumanNavigation cannot exclude every one. Checked BEFORE the
    // signup-session hand-off, so a session cookie cannot turn a GET into a
    // release either. The customer's TAP (POST chat=1 + this page's token,
    // same origin) claims the first hand-off.
    // A chat link that is already bound (credential released once) is NOT offered to a bare tap again: without the binding cookie the
    // browser falls through to the code sent to the customer's own WhatsApp below (the "Open my line" re-open; plan v1.9, 2026-10-15).
    if (begun.chatVerified && !begun.handedOff) return chatActivatePage(_req, code, begun.did)
    // ... and a bound chat link in a browser without the binding cookie gets a page that ASKS for a code - sent only on an explicit tap (POST),
    // never on a GET (link previews, prefetch, a visitor), and BEFORE the signup-session shortcut below so that cookie cannot turn a GET into a
    // release (Codex #231 P1, still in force; independent review 2026-10-15).
    if (begun.chatVerified) return chatReopenPage(_req, code, begun.did)
    if (sessionProvedNow(begun, sessionUserId)) {
      return handoff(code, begun, _req, { setCookie: true })
    }
    // Verified, but not by this browser: a forwarded link, or a browser that
    // lost its cookie. Fresh code, typed here, earns this browser its own
    // cookie. Never a dead end, never a credential without possession proof.
    const again = await beginActivation(code, true).catch(() => null)
    if (!again || again.state === 'expired') return activationUnavailablePage()
    return otpPage(_req, code, again.did)
  }
  return otpPage(_req, code, begun.did)
}

/** beginActivation() reports 'verified' for a signup-session match without a typed code; only then is a cookie-less verified GET this browser's own. */
function sessionProvedNow(begun: { userId?: string }, sessionUserId: string | null): boolean {
  return !!sessionUserId && !!begun.userId && sessionUserId === begun.userId
}

/**
 * The one place the credential is handed over. Repeat handoffs are what
 * make the link survive an install trip; the cookie is what keeps "repeat"
 * meaning "this browser" (lite-activation.ts, HANDOFF_COOKIE).
 */
async function handoff(
  code: string,
  begun: { sipUsername?: string; did?: string; userId?: string; nonceExpiresAt?: number },
  req: NextRequest,
  opts: { setCookie: boolean; firstBindOnly?: boolean },
): Promise<NextResponse> {
  if (!begun.sipUsername) return activationUnavailablePage()
  // firstBindOnly: only the atomic FIRST hand-off may succeed (chat-verified
  // links have no OTP to fall back on for a second browser).
  // chatAuthorized: every caller of handoff() has already authorised this
  // browser (typed code, signup session, bound cookie, or the chat tap POST).
  const h = await consumeForXml(code, begun.sipUsername, { repeat: !opts.firstBindOnly, chatAuthorized: true }).catch(() => null)
  if (!h) return opts.firstBindOnly ? chatLinkAlreadyOpenedPage() : activationUnavailablePage()
  // THE HAND-OFF NOTE (S5) goes out on the FIRST hand-off: "handed to Cloud
  // Softphone", never "set up" �?" a hand-off is not a phone (the owner's cold
  // walk, 2026-09-15). "Your line is live" is sent ONLY by the registration
  // watch on a real REGISTER (lib/lite-registration-followups.ts). Fire-and-
  // forget, never on the response path; the gate decides. Repeat handoffs
  // never re-send it: the first-handoff claim in consumeForXml() is atomic.
  if (h.firstHandoff && h.ownerPhone && begun.userId) {
    const ownerPhone = h.ownerPhone
    const userId = begun.userId
    void (async () => {
      try {
        const { sendHandoffNote, resolveLatestInboundRef } = await import('@/app/lib/lite-post-provision')
        const ref = await resolveLatestInboundRef(ownerPhone)
        // FROM the number the customer actually wrote to (the concierge
        // inbound row's metaPhoneId), else the number the concierge
        // intercepts (lite-concierge.ts CONCIERGE_PHONE_ID �?" the same value
        // the Flow follow-ups now send from, so a reply to this note routes
        // back to the concierge). GitHub Codex on PR #123, twice.
        const { CONCIERGE_PHONE_ID } = await import('@/app/lib/lite-concierge')
        const fromPhoneId = ref?.metaPhoneId || CONCIERGE_PHONE_ID
        await sendHandoffNote({
          phone: ownerPhone, userId,
          incomingPhoneId: fromPhoneId,
          gate: {
            purpose: 'conversation_reply', senderPath: 'go:handoff-note', userId,
            ...(ref
              ? { sourceEventType: ref.sourceEventType, sourceEventId: ref.sourceEventId }
              : { sourceEventType: 'lite_chat_verified_activation', sourceEventId: code }),
          },
        })
      } catch (e: any) {
        console.error('[go] hand-off note failed (the hand-off itself succeeded):', e?.message)
      }
    })()
  }
  // REGISTRATION WATCH accelerator: stamp `pending` now so the operator
  // screen shows the hand-off immediately, and look for the REGISTER ~90 s
  // out. Best effort only �?" a bff-v2 restart loses this timer and the cron
  // tick (lib/lite-registration-watch.ts) resumes from the activation row's
  // xmlFetchedAt regardless. Never on the response path.
  if (h.firstHandoff && h.liteAccountId) {
    const liteAccountId = h.liteAccountId
    try {
      void prisma.liteAccount
        .update({ where: { id: liteAccountId }, data: { registrationState: 'pending', registrationReason: null, registrationHandoffAt: new Date(), registrationCheckedAt: null } })
        .catch((e: any) => console.error('[go] registration pending stamp failed:', e?.message))
    } catch (e: any) { console.error('[go] registration pending stamp failed:', e?.message) }
    const t = setTimeout(() => {
      import('@/app/lib/lite-registration-watch')
        .then(({ checkOneAccount }) => checkOneAccount(liteAccountId))
        .catch((e: any) => console.error('[go] early registration check failed (cron will resume):', e?.message))
    }, 90_000)
    if (typeof (t as any).unref === 'function') (t as any).unref()
  }
  const res = launchHandoffPage(h, platformFromUa(req.headers.get('user-agent')), begun.did)
  if (opts.setCookie) {
    const value = handoffCookieValue(code)
    if (value) {
      res.cookies.set(HANDOFF_COOKIE, value, {
        httpOnly: true, secure: true, sameSite: 'lax',
        path: `/go/${encodeURIComponent(code)}`,
        maxAge: Math.max(1, Math.floor(((h.nonceExpiresAt ?? Date.now()) - Date.now()) / 1000)),
      })
    }
  }
  return res
}

/**
 * POST /go/<nonce> �?" the OTP form target (and `resend`). Possession proof:
 * the code was sent to the customer's OWN WhatsApp number, so entering it
 * demonstrates they hold that phone. A correct code goes STRAIGHT to the
 * hand-off page �?" no intermediate "which one are you" screen �?" and sets the
 * handoff cookie so this browser can come back for the credential again
 * within the link's TTL without a second code.
 *
 * `open=1` (the retired "Open my app" button) is still honoured for a tab
 * rendered before this change: with the cookie it hands off, without it the
 * browser is asked for a code.
 */
export async function POST(req: NextRequest, { params }: { params: Promise<{ code: string }> }) {
  const { code } = await params

  // DIAGNOSIS CORRECTED 2026-09-29 (def-app-signin-stuck-in-whatsapp-inapp-
  // browser-regression-2026-09-29): see the GET handler's comment above --
  // this POST is no longer diverted for in-app UAs either.

  const form = await req.formData().catch(() => null)
  const otp = String(form?.get('otp') ?? '').trim()
  const resend = String(form?.get('resend') ?? '') === '1'
  const open = String(form?.get('open') ?? '') === '1'
  const chat = String(form?.get('chat') ?? '') === '1'

  // CHAT-VERIFIED tap (Codex #231, P1): the explicit action that releases the
  // credential for a no-OTP link. Only a link the mint marked chat-verified
  // qualifies; anything else falls to the ordinary code path below.
  if (chat) {

    if (!chatTapTokenProves(String(form?.get('t') ?? ''), code) || !sameOriginPost(req)) return chatTapRefusedPage()
    const begun = await beginActivation(code, false, null).catch(() => null)
    if (!begun || begun.state === 'expired') return activationUnavailablePage()
    if (!(await activationEnabledFor(begun.userId ?? begun.sipUsername))) return activationUnavailablePage()
    if (begun.state === 'verified' && handoffCookieProves(req.cookies.get(HANDOFF_COOKIE)?.value, code)) {
      return handoff(code, begun, req, { setCookie: false })
    }
    if (begun.state === 'verified' && begun.chatVerified && !begun.handedOff) {
      return handoff(code, begun, req, { setCookie: true, firstBindOnly: true })
    }
    // Not a chat-verified link: exactly the ordinary code path.
    if (begun.state !== 'verified') return otpPage(req, code, begun.did)
    const again = await beginActivation(code, true).catch(() => null)
    if (!again || again.state === 'expired') return activationUnavailablePage()
    return otpPage(req, code, again.did)
  }

  if (open) {
    const begun = await beginActivation(code, false, null).catch(() => null)
    if (!begun || begun.state === 'expired') return activationUnavailablePage()
    if (!(await activationEnabledFor(begun.userId ?? begun.sipUsername))) return activationUnavailablePage()
    if (begun.state === 'verified' && handoffCookieProves(req.cookies.get(HANDOFF_COOKIE)?.value, code)) {
      return handoff(code, begun, req, { setCookie: false })
    }
    const again = await beginActivation(code, true).catch(() => null)
    if (!again || again.state === 'expired') return activationUnavailablePage()
    return otpPage(req, code, again.did)
  }

  if (resend) {
    const again = await beginActivation(code, true).catch(() => null)
    if (!again || again.state === 'expired') return activationUnavailablePage()
    if (!(await activationEnabledFor(again.userId ?? again.sipUsername))) return activationUnavailablePage()
    return otpPage(req, code, again.did, 'We sent a new code to your WhatsApp.')
  }

  const v = await verifyActivationOtp(code, otp).catch(() => null)
  if (!v) return activationUnavailablePage()

  if (v.result === 'ok') {
    if (!(await activationEnabledFor(v.userId ?? v.sipUsername))) return activationUnavailablePage()
    return handoff(code, v, req, { setCookie: true })
  }
  if (v.result === 'expired') return activationUnavailablePage()
  if (v.result === 'locked') {
    return otpPage(req, code, undefined, 'Too many incorrect codes. Request a new one below.')
  }
  return otpPage(req, code, undefined, 'That code is not right. Check WhatsApp and try again.')
}
